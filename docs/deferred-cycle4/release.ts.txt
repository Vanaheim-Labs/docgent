import { createHash } from "node:crypto";
import { cacheDriver, pdfStore } from "@/lib/pdf-cache";
import { renderMarkdown } from "@/lib/render";

/** Separate namespace: previews are never evidence of a completed release. */
export function releaseKey(brand: string, slug: string, content: string) {
  // Superseding retires an issue but must not change the PDF that was issued.
  const canonical = content.replace(/^(---\n)([\s\S]*?)(\n---\n)/, (_match, open, fm, close) =>
    open + fm.replace(/^status: superseded[ \t]*$/m, "status: released") + close);
  const digest = createHash("sha256").update(canonical).digest("hex");
  return `releases/${brand}/${slug}/${digest}.pdf`;
}

/** Archive before committing released status. Failed commits may leave harmless orphan artifacts. */
export async function archiveRelease(opts: {
  brand: string; slug: string; content: string; baseSha: string; git: any; docs: any;
}) {
  if (cacheDriver() !== "r2") throw new Error("Durable PDF archive is not configured.");
  const { brand, slug, content, baseSha, git, docs } = opts;
  const ref = await git.head();
  const inspected = await docs.readDocument(brand, slug, { ref });
  if (inspected.sha !== baseSha) throw new Error("Document changed; reload before release.");
  const dir = `documents/${slug}`;
  const tree = await git.tree({ ref, prefix: `${dir}/` });
  if (tree.truncated) throw new Error("Asset inventory is incomplete; release refused.");
  const assets: Record<string, string> = {};
  for (const entry of tree.entries) {
    if (entry.type !== "file" || !/^documents\/[^/]+\/(assets|figures)\//.test(entry.path)) continue;
    // GitStore's text reader cannot losslessly read binary assets. Fail closed
    // rather than issuing a corrupted PDF; ordinary previews remain available.
    if (!/\.svg$/i.test(entry.path)) throw new Error("Release currently supports SVG assets only; binary asset archiving requires a lossless reader.");
    const file = await git.readFile(entry.path, { ref });
    assets[entry.path.slice(dir.length + 1)] = Buffer.from(file.content, "utf8").toString("base64");
  }
  const { pdf } = await renderMarkdown(content, brand, assets);
  if (!pdf.subarray(0, 5).equals(Buffer.from("%PDF-"))) throw new Error("Renderer did not return a PDF.");
  const store = pdfStore();
  const key = releaseKey(brand, slug, content);
  await store.put(key, pdf);
  const persisted = await store.get(key);
  if (!persisted?.equals(pdf)) throw new Error("Archive readback failed or immutable artifact differs. Release refused.");
  return { key, sha256: createHash("sha256").update(pdf).digest("hex"), sourceCommit: ref };
}
